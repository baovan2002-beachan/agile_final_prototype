  // Tab Portal Switcher
  function switchPortal(portal) {
    const parentTab = document.getElementById('tab-parent');
    const staffTab = document.getElementById('tab-staff');
    const parentView = document.getElementById('parent-view');
    const staffView = document.getElementById('staff-view');
    const leftEyebrow = document.getElementById('left-eyebrow');
    const leftHeadline = document.getElementById('left-headline');
    const leftSubtext = document.getElementById('left-subtext');

    dismissBanner();

    if (portal === 'parent') {
      // Toggle Tabs
      parentTab.className = "w-1/2 py-space-xs font-label-md text-label-md rounded-lg text-center transition-all bg-surface-container-lowest text-primary shadow-sm font-semibold";
      staffTab.className = "w-1/2 py-space-xs font-label-md text-label-md rounded-lg text-center text-on-surface-variant hover:text-on-surface transition-all";
      
      // Toggle Views
      parentView.classList.remove('hidden');
      staffView.classList.add('hidden');

      // Update Left Hero Copy
      leftEyebrow.textContent = "STUDENT FINANCE PORTAL";
      leftHeadline.textContent = "Smarter student finances, made simple.";
      leftSubtext.textContent = "Empowering families and government stewards with transparent grants, seamless tuition deductions, and synchronized accounts.";
    } else {
      // Toggle Tabs
      staffTab.className = "w-1/2 py-space-xs font-label-md text-label-md rounded-lg text-center transition-all bg-surface-container-lowest text-secondary shadow-sm font-semibold";
      parentTab.className = "w-1/2 py-space-xs font-label-md text-label-md rounded-lg text-center text-on-surface-variant hover:text-on-surface transition-all";
      
      // Toggle Views
      parentView.classList.add('hidden');
      staffView.classList.remove('hidden');

      // Update Left Hero Copy
      leftEyebrow.textContent = "SCHOOL AND HQ ADMINISTRATION";
      leftHeadline.textContent = "Student finance records, kept in order.";
      leftSubtext.textContent = "Authoritative fiscal ledger access for school finance administrators and Ministry of Education auditing officers.";
    }
  }

  // Password Visibility Toggle
  function togglePasswordVisibility() {
    const passInput = document.getElementById('staff-password');
    const eyeIcon = document.getElementById('eye-icon');
    if (passInput.type === 'password') {
      passInput.type = 'text';
      eyeIcon.textContent = 'visibility_off';
    } else {
      passInput.type = 'password';
      eyeIcon.textContent = 'visibility';
    }
  }

  // Quick Role Auto-Fill
  function fillRole(role) {
    const emailField = document.getElementById('staff-email');
    const passField = document.getElementById('staff-password');
    if (role === 'school') {
      emailField.value = 'alicia.tan@riverside.edu.sg';
      passField.value = '••••••••••••';
      showBanner("Pre-filled credentials for Alicia Tan (School Admin)", "info");
    } else if (role === 'moe') {
      emailField.value = 'daniel.lim@moe.gov.sg';
      passField.value = '••••••••••••';
      showBanner("Pre-filled credentials for Daniel Lim (MOE HQ)", "info");
    }
  }

  // Staff Sign-in Form Submit Handler
  function handleStaffSubmit(e) {
    e.preventDefault();
    const email = document.getElementById('staff-email').value;
    const password = document.getElementById('staff-password').value;

    if (password === 'wrong' || password === '1234') {
      showBanner("The email or password is incorrect. Please verify institutional credentials.", "error");
    } else {
      showBanner("Authenticating institutional session...", "info");
      setTimeout(() => {
        if (email.includes('moe.gov.sg')) {
          alert('Welcome Officer Daniel Lim. Redirecting to MOE HQ Treasury Dashboard.');
        } else {
          alert('Welcome Alicia Tan. Redirecting to School Administration Portal.');
        }
      }, 700);
    }
  }

  // Singpass Modal Management
  function openSingpassModal() {
    document.getElementById('singpass-modal').classList.remove('hidden');
    document.getElementById('sp-stage-ready').classList.remove('hidden');
    document.getElementById('sp-stage-verifying').classList.add('hidden');
    document.getElementById('sp-stage-success').classList.add('hidden');

  // View switcher function for all 11 views
  function navigateView(viewName) {
    // Hide all view sections
    document.querySelectorAll('.view-section').forEach(el => {
      el.classList.add('hidden');
      el.classList.remove('flex');
    });

    // Show targeted view

    if (target) {

      target.classList.add('flex');

    // Update sidebar navigation active status

        link.setAttribute('aria-current', 'page');
      }

  // School filtering
  function filterSchools() {
    const query = (document.getElementById('school-search') ? document.getElementById('school-search').value.toLowerCase() : '');
    const zone = (document.getElementById('zone-filter') ? document.getElementById('zone-filter').value : 'ALL');
    const type = (document.getElementById('type-filter') ? document.getElementById('type-filter').value : 'ALL');
    const rows = document.querySelectorAll('.school-row');

    let visibleCount = 0;
    rows.forEach(row => {
      const name = (row.getAttribute('data-name') || '').toLowerCase();
      const code = (row.getAttribute('data-code') || '').toLowerCase();
      const rowZone = row.getAttribute('data-zone') || '';
      const rowType = row.getAttribute('data-type') || '';

      const matchesQuery = name.includes(query) || code.includes(query);
      const matchesZone = (zone === 'ALL' || rowZone === zone);
      const matchesType = (type === 'ALL' || rowType === type);

      if (matchesQuery && matchesZone && matchesType) {
        row.style.display = '';
        visibleCount++;
      } else {
        row.style.display = 'none';
      }
    });

    const countEl = document.getElementById('school-count');
    if (countEl) countEl.innerText = visibleCount;
  }
